Contract this plan CREATES:

```typescript
// interactable-def.ts
import type Phaser from "phaser";
export interface InteractableDef {
  id: string;                                   // equals its registry key
  place(scene: Phaser.Scene, anchor: { x: number; y: number }): Phaser.GameObjects.GameObject; // returns the interactive root (already setInteractive with a hit area)
  onClick(scene: Phaser.Scene, root: Phaser.GameObjects.GameObject): void; // visual reaction only
  lines?: readonly string[];                    // mascot only: 3-5 tip/joke lines
}
// registry.ts
export const INTERACTABLE_REGISTRY = { campfire, fireflies, lantern, mascot } satisfies Readonly<Record<string, InteractableDef>>;
export type InteractableId = keyof typeof INTERACTABLE_REGISTRY;
```
