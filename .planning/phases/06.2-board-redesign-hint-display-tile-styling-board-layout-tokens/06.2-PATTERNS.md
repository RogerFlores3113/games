### `packages/rules/src/hanabi/legality.ts` — `canReorderDiscard` (service, CRUD)

**Analog:** `canReorder`, same file, lines 106-128.

```typescript
// Source: packages/rules/src/hanabi/legality.ts:106-128
export function canReorder(
  state: HanabiState,
  actorSeatId: string,
  cardIds: readonly string[],
): Legality {
  if (isGameOver(state)) return { legal: false, reason: "game_over" };
  const hand = state.hands.find((h) => h.seatId === actorSeatId);
  if (hand === undefined) return { legal: false, reason: "card_not_in_hand" };
  const currentIds = hand.slots.map((s) => s.card.id);
  if (cardIds.length !== currentIds.length) {
    return { legal: false, reason: "card_not_in_hand" };
  }
  const currentSet = new Set(currentIds);
  const suppliedSet = new Set(cardIds);
  if (currentSet.size !== suppliedSet.size) {
    return { legal: false, reason: "card_not_in_hand" };
  }
  for (const id of cardIds) {
    if (!currentSet.has(id)) return { legal: false, reason: "card_not_in_hand" };
  }
  return { legal: true };
}
```

**Copy-and-adapt for `canReorderDiscard`:**
- Same `isGameOver` gate, same permutation-check structure (length match, set-size match, membership loop).
- Replace `hand.slots.map((s) => s.card.id)` with `state.discard.map((c) => c.id)` — no per-seat hand lookup at all (D-25: any seated player).
- Add an explicit `state.seatIds.includes(actorSeatId)` check as the not-found branch (there is no "hand === undefined" fallback here since discard is shared, not per-seat).
- Reuse the exact same `reason` string style (`"card_not_in_hand"` is the closest existing reason literal; confirm at plan time whether a new reason like `"card_not_in_discard"` is warranted for clarity — RESEARCH.md does not mandate reusing the literal, only the *check shape*).

export function isReorderRequest(
  request: unknown,
): request is { type: "reorder"; cardIds: string[] } {
  if (typeof request !== "object" || request === null) return false;
  const keys = Object.keys(request);
  if (keys.length !== 2 || !keys.includes("type") || !keys.includes("cardIds")) return false;
  const r = request as { type: unknown; cardIds: unknown };
  if (r.type !== "reorder" || !Array.isArray(r.cardIds)) return false;
  return r.cardIds.every((id) => typeof id === "string");
}

```typescript
// Source: packages/rules/src/hanabi/actions.ts:421-453
function applyReorder(
  state: HanabiState,
  actorSeatId: string,
  cardIds: readonly string[],
): AdapterResult<HanabiState> {
  const legality = canReorder(state, actorSeatId, cardIds);
  if (!legality.legal) return { ok: false, error: legality.reason };

  const hand = state.hands.find((h) => h.seatId === actorSeatId)!;
  const slotsById = new Map(hand.slots.map((s) => [s.card.id, s]));
  const reorderedSlots = cardIds.map((id) => slotsById.get(id)!);

  const hands = state.hands.map((h) =>
    h.seatId === actorSeatId ? { seatId: h.seatId, slots: reorderedSlots } : h,
  );

  return {
    ok: true,
    state: {
      variant: state.variant,
      seatIds: state.seatIds,
      turnIndex: state.turnIndex,
      hands,
      deck: state.deck,
      stacks: state.stacks,
      discard: state.discard,
      clueTokens: state.clueTokens,
      fuses: state.fuses,
      finalTurnsRemaining: state.finalTurnsRemaining,
      history: state.history,
    },
  };
}
```

**Copy-and-adapt for `applyReorderDiscard`:** validate via `canReorderDiscard`, set `discardOrder: cardIds` instead of remapping `hands`, and return the **full field-by-field state literal exactly like this** — every field the existing literal lists, plus the new `discardOrder` field. This project's error-handling discipline is: no spread, no `Object.assign`, no omit-helper — a missing field is a `tsc` error, which is the intended catch mechanism (per RESEARCH.md's explicit anti-pattern warning).

**Critical ripple:** every OTHER action branch in this file (`applyPlay`, `applyDiscard`, `applyClue`, and `applyReorder` itself) returns a full state literal in the same shape (see `applyDiscard`, lines 295-345+, and `applyReorder` above) — **all of them must add `discardOrder: state.discardOrder` (or the newly-appended value for `applyDiscard`) to their own return literal**, or `tsc` will fail to compile. `applyDiscard` specifically must append the newly-discarded card's id to the end of `discardOrder` (mirrors its existing `const discard = [...state.discard, card];` on line 311) — this is the one branch where `discardOrder` is not merely "carried forward unchanged."

**Dispatch:** add `if (action.type === "reorderDiscard") return applyReorderDiscard(state, actorSeatId, action.cardIds);` beside the existing dispatch line (`actions.ts:469`).

```typescript
// Source: apps/web/lib/hanabi-drag-logic.ts:81-95
export function reorderedCardIds(
  handIds: readonly string[],
  draggedId: string,
  targetIndex: number,
): string[] {
  const currentIndex = handIds.indexOf(draggedId);
  if (currentIndex === -1) {
    return [...handIds];
  }
  const without = handIds.filter((id) => id !== draggedId);
  const clampedIndex = Math.max(0, Math.min(targetIndex, without.length));
  const result = without.slice();
  result.splice(clampedIndex, 0, draggedId);
  return result;
}
```
This function is directly reusable/mirrorable for discard-pile ids — same remove-then-insert move semantics, no swap.

```typescript
// Source: apps/web/lib/hanabi-drag-logic.ts:117-137
export function applyPendingOrder<T extends { id: string }>(
  cards: T[],
  pendingIds: readonly string[] | null,
): T[] {
  if (pendingIds === null) return cards;
  if (pendingIds.length !== cards.length) return cards;
  const byId = new Map(cards.map((card) => [card.id, card]));
  const result: T[] = [];
  for (const id of pendingIds) {
    const card = byId.get(id);
    if (card === undefined) return cards;
    result.push(card);
  }
  return result;
}
```
Reuse verbatim (generic over `{ id: string }`, already works for discard cards) for the optimistic-order-before-server-confirmation behavior (D-26/D-27).

**Error handling / stale-submission pattern (from `requestForDrop`, lines 142-169):** a discard-pile equivalent should build a `DropRequest`-shaped `{ type: "reorderDiscard", cardIds }` object, returning `null` for a no-op drop (unchanged order, "none" target) exactly like the existing function's early returns. Do NOT re-derive legality here — dispatch through the same `act()` chokepoint and let `canReorderDiscard`'s permutation check on the server reject a stale submission (D-26 is satisfied by that rejection, not by new client logic).
