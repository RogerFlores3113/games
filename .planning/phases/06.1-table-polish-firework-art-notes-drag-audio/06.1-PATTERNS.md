export function isPlayRequest(request: unknown): request is { type: "play"; cardId: string } {

function drawCard(input: {
  config: ReturnType<typeof variantConfig>;
  seatId: string;
  hands: readonly Hand[];
  deck: readonly HanabiCard[];
  finalTurnsRemainingBeforeThisTurn: number | null;
}): { hands: Hand[]; deck: readonly HanabiCard[]; drawnCardId: string | null } {
  const { config, seatId, hands, deck, finalTurnsRemainingBeforeThisTurn } = input;
  if (finalTurnsRemainingBeforeThisTurn !== null || deck.length === 0) {
    return { hands: hands.map((h) => h), deck, drawnCardId: null };

  const drawnCard = deck[0]!;

  const nextHands = hands.map((hand) => {

function removeFromHand(
  hands: readonly Hand[],
  actorSeatId: string,
  cardId: string,
): Hand[] {
  return hands.map((hand) => {
    if (hand.seatId !== actorSeatId) return hand;

}

export function canPlay(state: HanabiState, actorSeatId: string, cardId: string): Legality {

function getLocalStorage(): Storage | undefined {
  if (typeof window === "undefined") {
    return undefined;
  }

export function readSeatToken(code: string): string | undefined {

  if (!storage) {
    return undefined;
  }
  try {

  } catch {
    return undefined;
  }
}

### `apps/web/lib/hanabi-visual-logic.ts` — reused verbatim for stack-completion detection (utility, transform) — shared pattern for audio cues

**Analog/source of truth:** same file, `newlyCompletedStacks` (lines 100-115)

```typescript
export function newlyCompletedStacks(
  prev: HanabiView["stacks"],
  next: HanabiView["stacks"],
): Suit[] {
  const maxRank = RANKS[RANKS.length - 1];
  const prevBySuit = new Map(prev.map((entry) => [entry.suit, entry.topRank]));
  const completed: Suit[] = [];
  for (const entry of next) {
    if (entry.topRank !== maxRank) continue;
    const prevRank = prevBySuit.get(entry.suit) ?? 0;
    if (prevRank < maxRank) {
      completed.push(entry.suit);
    }
  }
  return completed;
}
```
`hanabi-audio-cues.ts`'s D-28 "completion-only sound" branch MUST call this exact function (already used by `Table.tsx`'s visual flash) rather than re-deriving "did this play complete a stack" a second way — per RESEARCH.md's "Don't Hand-Roll" table, this is the single named risk of this phase's audio work.
